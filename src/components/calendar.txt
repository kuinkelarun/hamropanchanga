        const minBsYear = parseInt(Object.keys(bsCalendarData)[0]);
        const maxBsYear = parseInt(Object.keys(bsCalendarData)[Object.keys(bsCalendarData).length - 1]);

        /**
         * Converts a date from AD (English) to BS (Nepali) format using the new data.
         * @param {number} year - The AD year.
         * @param {number} month - The AD month (0-11).
         * @param {number} day - The AD day.
         * @returns {{year: number, month: number, day: number, dayOfWeek: number}} The BS date and day of the week.
         */
        function convertAdToBs(year, month, day) {
            const adDate = new Date(year, month, day);
            
            let bsYear = 0;
            let totalDays = 0;

            for (const bsY in bsCalendarData) {
                const startAdDate = bsCalendarData[bsY].startAdDate;
                if (adDate >= startAdDate) {
                    bsYear = parseInt(bsY);
                    totalDays = Math.floor((adDate - startAdDate) / (1000 * 60 * 60 * 24)) + 1;
                } else {
                    break;
                }
            }

            let bsMonth = 0;
            let bsDay = totalDays;
            
            for (let i = 0; i < bsCalendarData[bsYear].daysInMonths.length; i++) {
                const daysInMonth = bsCalendarData[bsYear].daysInMonths[i];
                if (bsDay <= daysInMonth) {
                    bsMonth = i + 1;
                    break;
                }
                bsDay -= daysInMonth;
            }
            
            return { year: bsYear, month: bsMonth, day: bsDay, dayOfWeek: adDate.getDay() };
        }

        /**
         * Converts a date from BS (Nepali) to AD (English) format using the new data.
         * @param {number} year - The BS year.
         * @param {number} month - The BS month (1-12).
         * @param {number} day - The BS day.
         * @returns {{year: number, month: number, day: number}} The AD date.
         */
        function convertBsToAd(year, month, day) {
            const startAdDate = bsCalendarData[year].startAdDate;
            let totalDays = 0;
            for (let i = 0; i < month - 1; i++) {
                totalDays += bsCalendarData[year].daysInMonths[i];
            }
            totalDays += day - 1;

            const adDate = new Date(startAdDate);
            adDate.setDate(startAdDate.getDate() + totalDays);
            return { year: adDate.getFullYear(), month: adDate.getMonth(), day: adDate.getDate() };
        }
        
        /**
         * Gets the number of days in a given BS month.
         * @param {number} year - The BS year.
         * @param {number} month - The BS month (1-12).
         * @returns {number} The number of days in the month.
         */
        function getDaysInBsMonth(year, month) {
            if (!bsCalendarData[year] || month < 1 || month > 12) {
                return 0;
            }
            return bsCalendarData[year].daysInMonths[month - 1];
        }

        /**
         * Converts a number to Nepali script.
         * @param {number} num - The number to convert.
         * @returns {string} The Nepali number string.
         */
        function toNepaliNumber(num) {
            return String(num).split('').map(digit => nepaliNumbers[parseInt(digit)]).join('');
        }

        /**
         * Adjusts the current date to Nepal Standard Time (NPT).
         * NPT is UTC +5:45.
         */
        function getNepalDate() {
            const now = new Date();
            const utc = now.getTime() + (now.getTimezoneOffset() * 60000);
            const nptOffset = 5.75 * 3600000;
            return new Date(utc + nptOffset);
        }

                // English Date (Secondary)
                const englishDateElement = document.createElement('div');
                englishDateElement.className = 'absolute bottom-2 right-2 text-sm font-medium text-gray-400';
                englishDateElement.textContent = adDate.day;
                dayTile.appendChild(englishDateElement);

                // Tithi Edit Button (Vertical dots)
                const editButton = document.createElement('button');
                editButton.className = 'edit-btn';
                editButton.innerHTML = '&#x22EE;'; // U+22EE Vertical Ellipsis
                editButton.title = 'Edit Tithis/Events';
                editButton.addEventListener('click', (e) => {
                    e.stopPropagation();
                    openModal(dayTile.dataset.dateKey);
                });
                dayTile.appendChild(editButton);

                // Tithi list display (Summary)
                const tithiDisplay = document.createElement('div');
                tithiDisplay.className = 'tithi-list-display pt-8';
                dayTile.appendChild(tithiDisplay);

                // Events list display (Summary)
                const eventDisplay = document.createElement('div');
                eventDisplay.className = 'event-list-display';
                dayTile.appendChild(eventDisplay);

                renderSummariesForDay(dateKey, tithiDisplay, eventDisplay);
                
                // --- Updated click event logic to handle the user's request ---
                dayTile.addEventListener('click', (e) => {
                    e.stopPropagation(); 
                    // If a popup is already open and it's not the one we are clicking on, close it and prevent a new one from opening.
                    // This requires a second click to open the new popup, as requested by the user.
                    if (activeDayTile && activeDayTile !== dayTile) {
                        hidePopup();
                    } else {
                        // If no popup is open or it's the same tile, toggle the popup.
                        if (activeDayTile === dayTile) {
                            hidePopup();
                        } else {
                            showPopup(dayTile, dateKey);
                        }
                    }
                });
                // --- End of updated logic ---
                
                calendarGrid.appendChild(dayTile);
            }
        }

        function handleOutsideClick(e) {
            // Check if the click is outside the popup and the currently active day tile.
            // This ensures that clicking on other day tiles is handled by the tile's own event listener, not this document-wide listener.
            if (!dayPopup.contains(e.target) && !activeDayTile.contains(e.target)) {
                hidePopup();
            }
        }

        function showPopup(dayTile, dateKey) {
            hidePopup(); // Always start with a clean slate
            
            activeDayTile = dayTile;
            
            const [adYear, adMonth, adDay] = dateKey.split('-').map(Number);
            
            const adDate = new Date(adYear, adMonth - 1, adDay);
            const bsDate = convertAdToBs(adDate.getFullYear(), adDate.getMonth(), adDate.getDate());

            const popupNepaliDate = document.getElementById('popup-nepali-date');
            const popupEnglishDate = document.getElementById('popup-english-date');
            const popupTithis = document.getElementById('popup-tithis');
            const popupEvents = document.getElementById('popup-events');
            
            popupNepaliDate.textContent = `${toNepaliNumber(bsDate.day)} ${nepaliMonths[bsDate.month - 1]}, ${toNepaliNumber(bsDate.year)}`;
            popupEnglishDate.textContent = `${englishMonths[adDate.getMonth()]} ${adDate.getDate()}, ${adDate.getFullYear()}`;

            const tithis = tithisByDate[dateKey] || [];
            popupTithis.innerHTML = tithis.length > 0 ? tithis.map(t => `<p class="text-sm"><span class="font-medium">${t.name}</span> <span class="text-gray-500 text-xs">(${t.startTime}-${t.endTime})</span></p>`).join('') : '<p class="text-gray-400 text-xs">No tithis.</p>';

            const events = eventsByDate[dateKey] || [];
            popupEvents.innerHTML = events.length > 0 ? events.map(event => `<p class="text-sm font-medium">${event.name}</p>`).join('') : '<p class="text-gray-400 text-xs">No events.</p>';

            const rect = dayTile.getBoundingClientRect();
            
            // Show the popup temporarily to get its dimensions
            dayPopup.style.display = 'flex';
            const popupRect = dayPopup.getBoundingClientRect();

            // Calculate space above and below the day tile
            const spaceAbove = rect.top;
            const spaceBelow = window.innerHeight - rect.bottom;

            // Positioning logic: prefer below, fallback to above
            if (spaceBelow >= popupRect.height) {
                dayPopup.style.top = `${rect.bottom + window.scrollY}px`;
                dayPopup.style.bottom = 'auto';
            } else if (spaceAbove >= popupRect.height) {
                dayPopup.style.top = `${rect.top + window.scrollY - popupRect.height}px`;
                dayPopup.style.bottom = 'auto';
            } else {
                 // Fallback to positioning below and adjust scroll if needed.
                dayPopup.style.top = `${rect.bottom + window.scrollY}px`;
                dayPopup.style.bottom = 'auto';
            }

            // Adjust horizontal position to keep popup on screen
            let leftPosition = rect.left + window.scrollX;
            if (leftPosition + popupRect.width > window.innerWidth - 10) {
                leftPosition = window.innerWidth - popupRect.width - 10;
            }
            dayPopup.style.left = `${leftPosition}px`;
            
            dayPopup.classList.add('open');

            document.addEventListener('click', handleOutsideClick);
        }

        function hidePopup() {
            if (!activeDayTile) return;

            dayPopup.classList.remove('open');
            setTimeout(() => {
                dayPopup.style.display = 'none';
            }, 200);

            document.removeEventListener('click', handleOutsideClick);
            activeDayTile = null;
        }

        function toEnglishNumber(nepaliNum) {
            const map = {
                '०': '0', '१': '1', '२': '2', '३': '3', '४': '4',
                '५': '5', '६': '6', '७': '7', '८': '8', '९': '9'
            };
            return parseInt(nepaliNum.split('').map(char => map[char]).join(''));
        }

        function openModal(dateKey) {
            hidePopup(); // Close popup before opening modal
            currentEditingDate = dateKey;
            
            const [adYear, adMonth, adDay] = dateKey.split('-').map(Number);
            const adDate = new Date(adYear, adMonth - 1, adDay);

            const bsDate = convertAdToBs(adDate.getFullYear(), adDate.getMonth(), adDate.getDate());
            const dayOfWeekNepali = nepaliWeekdays[bsDate.dayOfWeek];
            
            const formattedNepaliDate = `${toNepaliNumber(bsDate.day)} ${nepaliMonths[bsDate.month - 1]} ${toNepaliNumber(bsDate.year)}, ${dayOfWeekNepali}`;
            const formattedEnglishDate = `${englishMonths[adDate.getMonth()]} ${adDate.getDate()}, ${adDate.getFullYear()}`;
            const formattedHeader = `${formattedNepaliDate}<br><span class="text-base text-gray-500 font-normal">${formattedEnglishDate}</span>`;
            
            modalDateDisplay.innerHTML = formattedHeader;
            
            validationMessage.style.display = 'none';
            
            renderItemsInModal();
            tithiModal.style.display = 'flex';
            setTimeout(() => tithiModal.classList.add('open'), 10);
        }

        function closeModal() {
            tithiModal.classList.remove('open');
            setTimeout(() => {
                tithiModal.style.display = 'none';
            }, 300);
        }

        function renderItemsInModal() {
            tithiListModal.innerHTML = '';
            const tithis = tithisByDate[currentEditingDate] || [];
            const events = eventsByDate[currentEditingDate] || [];
            
            tithiListModal.innerHTML = `
                <h5 class="font-bold text-sm text-gray-800 mb-2">Tithis</h5>
                ${tithis.length > 0 ? tithis.map(t => `
                    <div class="p-3 mb-2 bg-gray-100 rounded-md flex justify-between items-center">
                        <div>
                            <p class="font-medium">${t.name}</p>
                            <p class="text-xs text-gray-600">${t.startTime} - ${t.endTime}</p>
                        </div>
                        <button class="delete-btn text-red-500 hover:text-red-700" data-id="${t.id}" data-type="tithi">&times;</button>
                    </div>`).join('') : '<p class="text-gray-500 text-sm">No tithis added yet.</p>'}

                <h5 class="font-bold text-sm text-gray-800 mt-4 mb-2">Events</h5>
                ${events.length > 0 ? events.map(e => `
                    <div class="p-3 mb-2 bg-gray-100 rounded-md flex justify-between items-center">
                        <div>
                            <p class="font-medium">${e.name}</p>
                        </div>
                        <button class="delete-btn text-red-500 hover:text-red-700" data-id="${e.id}" data-type="event">&times;</button>
                    </div>`).join('') : '<p class="text-gray-500 text-xs">No events added yet.</p>'}
            `;
            
            document.querySelectorAll('.delete-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    const id = e.target.dataset.id;
                    const type = e.target.dataset.type;
                    if (type === 'tithi') {
                        deleteFromDb('tithis', id);
                    } else if (type === 'event') {
                        deleteFromDb('events', id);
                    }
                });
            });
        }

        function renderSummariesForDay(dateKey, tithiElement, eventElement) {
            const tithis = tithisByDate[dateKey] || [];
            const events = eventsByDate[dateKey] || [];
            
            tithiElement.innerHTML = tithis.map(tithi => `<p class="text-xs text-gray-600 truncate">${tithi.name}</p>`).join('');
            eventElement.innerHTML = events.map(event => `<p class="text-xs text-gray-600 truncate">${event.name}</p>`).join('');
        }

        // Database Operations
        async function addToDb(collectionName, data) {
            try {
                const collectionRef = collection(db, `artifacts/${appId}/users/${userId}/${collectionName}`);
                await addDoc(collectionRef, data);
                renderItemsInModal();
            } catch (e) {
                console.error(`Error adding document to ${collectionName}: `, e);
            }
        }

        async function deleteFromDb(collectionName, docId) {
            try {
                const docRef = doc(db, `artifacts/${appId}/users/${userId}/${collectionName}`, docId);
                await deleteDoc(docRef);
            } catch (e) {
                console.error(`Error deleting document from ${collectionName}: `, e);
            }
        }

        // Event Listeners
        addTithiEventBtn.addEventListener('click', () => {
            const newTithi = tithiSelect.value;
            const newEvent = eventNameInput.value;
            const startTime = startTimeInput.value;
            const endTime = endTimeInput.value;

            // Clear previous validation messages
            validationMessage.style.display = 'none';

            // Validation for Tithi/Time entries
            if (newTithi && (startTime || endTime)) {
                if (!startTime || !endTime) {
                    validationMessage.textContent = 'Both start and end times are required for a Tithi.';
                    validationMessage.style.display = 'block';
                    return;
                }
                
                if (endTime <= startTime) {
                    validationMessage.textContent = 'End time must be after the start time.';
                    validationMessage.style.display = 'block';
                    return;
                }

                addToDb('tithis', { date: currentEditingDate, name: newTithi, startTime, endTime });
                tithiSelect.value = '';
                startTimeInput.value = '';
                endTimeInput.value = '';

            }
            
            // Validation for Event entries
            if (newEvent) {
                 addToDb('events', { date: currentEditingDate, name: newEvent });
                 eventNameInput.value = '';
            }

            if (!newTithi && !newEvent) {
                validationMessage.textContent = 'Please enter a Tithi or an Event.';
                validationMessage.style.display = 'block';
            }
        });

        closeModalBtn.addEventListener('click', closeModal);
        tithiModal.addEventListener('click', (e) => {
            if (e.target.id === 'tithi-modal') {
                closeModal();
            }
        });

        prevMonthBtn.addEventListener('click', () => {
            currentBsMonth--;
            if (currentBsMonth < 1) {
                currentBsMonth = 12;
                currentBsYear--;
            }
            renderCalendar();
        });

        nextMonthBtn.addEventListener('click', () => {
            currentBsMonth++;
            if (currentBsMonth > 12) {
                currentBsMonth = 1;
                currentBsYear++;
            }
            renderCalendar();
        });

        yearSelect.addEventListener('change', (e) => {
            currentBsYear = parseInt(e.target.value);
            renderCalendar();
        });
